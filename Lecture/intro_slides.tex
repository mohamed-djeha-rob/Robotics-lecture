\begin{frame}{Robotics: profound historical and cultural roots}
	Isaac Asimov (1940s), author of \emph{The Robot} roman, introduced robots as mechanical artifacts of human appearance but devoid of feelings.
	
	\begin{minipage}{0.65\columnwidth}
		 \textbf{Asimov's three fundamental laws:}
		\begin{enumerate}
			\item A robot may not injure human beings 
			\item A robot must obey the orders given by human beings, except when such orders would conflict with the first law
			\item A robot must protect its own existence, as long as this protection does not conflict with the first and second laws
		\end{enumerate}
	\end{minipage}

\begin{frame}{Robotics \& Robot}
\begin{tcolorbox}[colback=blue!5!white, colframe=green!50!black, title=Definition]
	\begin{itemize}
		\item Robotics is concerned with the study of the robots that can replace human beings in the execution of a task, as regards both physical activity and decision making
		\item A robot can be seen as a machine that is able to modify and interact with its environment in which it operates while respecting the Asimov's rules
	\end{itemize}

\begin{frame}{Robotic manipulator - Structure}
	\begin{itemize}
		\item \textbf{Robotic manipulator:} sequence of bodies (links) connected through articulation (joints).
		\item The rigid body tree (kinematic chain) is constituted by the \emph{arm}, the \emph{wrist} and the end-effector (hand)
		\item Two types of kinematic chain
		\begin{itemize}
			\item \emph{Open kinematic chain:} when there is only one kinematic chain between two ends. It can be a simple chain (robotic arm) or a tree kinematic chain (dual-arm robot, humanoid, quadruped, etc.) 
			\item \emph{Closed kinematic chain:} when a sequence of links forms a loop
		\end{itemize}

\begin{frame}{Robotic manipulator - Joints}
		\begin{itemize}
		\item Basic joints are \emph{prismatic}, \emph{revolute} 
		\item Prismatic joint allows a relative translation between two bodies 
		\item Revolute joint produces a relative rotation between two bodies
	\end{itemize}
	\begin{figure}
		\centering
		\includegraphics[width=0.6\columnwidth]{joints_type}
		\caption{Conventional representation of revloute and prismatic joints}
	\end{figure}
\end{frame}
\begin{frame}{Robotic manipulator - Joints}
	\begin{itemize}
		\item Any other types of joints can be modeled through prismatic or revolute joints, e.g., 
		\begin{itemize}
			\item Spherical joint: 3 revolute joints with concurrent axes (agile eye)
			\item Helical joint: combination of a translation $x$ and a rotation $\theta$ along the same axis: $x=p\theta$
		\end{itemize}
		\item Every allowed motion by a joint is a Degree of Freedom (DoF)
		\begin{itemize}
			\item Revolute, prismatic and helical joints: 1 DoF
			\item Spherical joint: 3 DoF
		\end{itemize}
		\item Each joint $\vectorQ$ has mechanical limits ($\vectorQ_{\min}$, $\vectorQ_{\max}$), velocity limits ($\vectorQdot_{\min}$, $\vectorQdot_{\max}$) and acceleration limits  ($\vectorQddot_{\min}$, $\vectorQddot_{\max}$)
		\item In practice, other types of joints can be used for modeling purposes, e.g., fixed (0 DoF) and floating joints (6 DoF)
	\end{itemize}
\end{frame}
\begin{frame}{Robotic manipulator - Joint-space and Task-space}
	\begin{itemize}
		\item \textbf{Joint-space:} space in which the joint configuration $\vectorQ$ is defined. 
		\item For a robotic arm with $n$ joints, the joint-space $\subseteq\mathbb{R}^n$ 
		\item \textbf{Task-space:} space in which the end-effector configuration $\vectorX_{ee}$ is defined.
		\item  $\vectorX_{ee}$ can be the position, the orientation, or both 
	\end{itemize} 
\end{frame}
\begin{frame}{Robotic manipulator - Workspace}
	\begin{itemize}
	%	\item Degree of Freedom (DoF) = number of possible motion - constraints imposed by the joint $\Rightarrow$ Each revolute or prismatic joint allows one DoF 
		\item \textbf{Workspace:} space reachable by the end-effector. Its shape and volume are defined by the robot structure and the mechanical joint limits. 
		\begin{figure}
			\centering
			\includegraphics[width = 0.22\columnwidth]{Cartesian robot.png}
			\includegraphics[width = 0.22\columnwidth]{Cylindrical robot.png}
			\includegraphics[width = 0.22\columnwidth]{Spherical robot.png}
			\includegraphics[width = 0.21\columnwidth]{anthropomorphic.png}
			\caption{Types of robots according to the workspace shape: \textbf{Cartesian} (left), \textbf{cylindrical} (left-center), \textbf{spherical} (right-center) an \textbf{anthropomorphic} (right)}
		\end{figure}
	\end{itemize}
\end{frame}


\begin{frame}{Robotic manipulator - Repeatability and Payload}
	\begin{itemize}
		\item \textbf{Repeatability:} measure of the robot ability to return to a previously reached position. In general, it is less than $0.1~mm$
		\begin{figure}
			\centering
			\includegraphics[height=0.45\textheight]{repetability}
		\end{figure}
		\item \textbf{Payload:} maximum load the robot can widthstand statically.
	\end{itemize} 
\end{frame}
