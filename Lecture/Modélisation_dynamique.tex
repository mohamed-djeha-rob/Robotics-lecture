\begin{frame}{Dynamic model}
	\begin{itemize}
		\item The dynamic model describes how joint torques $\vectorTau$ is related ($\vectorQ,\vectorQdot, \vectorQddot$) while accounting for all physical effects: links inertia and center of mass, Coriolis/centrifugal forces, gravity, friction, and external forces
		\item According to this relation, there exists two dynamic models
			\begin{figure}
			\centering
			\begin{tikzpicture}[auto, node distance=1.3cm, >=latex]
				
				\tikzstyle{block} = [draw, rectangle, minimum width=3cm, minimum height=1cm];
				
				% Nodes
				\node[block] (b1) {Direct dynamic model};
				\node[block, below of=b1] (b2) {Inverse  dynamic model};

				
				% Inputs
				\draw[->] ($(b1.west)+(-1,0)$) -- (b1.west) node[midway, above] {$\vectorTau$};
				\draw[->] ($(b2.west)+(-1,0)$) -- (b2.west)node[pos=0.1, above] {$\vectorQ,\vectorQdot, \vectorQddot$};

				
				% Outputs
				\draw[->] (b1.east) -- ($(b1.east)+(1,0)$)node[pos=0.9, above] {$\vectorQ,\vectorQdot, \vectorQddot$};
				\draw[->] (b2.east) -- ($(b2.east)+(1,0)$)node[midway, above] {$\vectorTau$};
				
			\end{tikzpicture}
		\end{figure}
	\end{itemize}
	
\end{frame}
