\begin{frame}{Forward differential kinematic model}
	Forward differential kinematic model allows to compute the task-space velocity $\vectorXdot_{ee}$ based on the joint velocity $\vectorQdot$
	\begin{equation*}
		\vectorXdot_{ee}=\matriceJ(\vectorQ)\vectorQdot, \ \vectorQdot\inR^n 
	\end{equation*}
	\begin{itemize}
		\item $\matriceJ\inR^{m\times n}$ is the Jacobian matrix, and $\vectorXdot_{ee}=\begin{bmatrix}
			\vectorV_{ee}\\\vectorOmega_{ee}
		\end{bmatrix}\inR^6$ with $\vectorV_{ee}$ is the linear end-effector velocity  and  $\vectorOmega_{ee}$ is the angular end-effector velocity, both w.r.t the base frame
		\item Jacobian $\matriceJ$ can be decomposed into $\matriceJ =  \begin{bmatrix}
			\matriceJ_{lin}\\\matriceJ_{rot}
		\end{bmatrix}$
	\end{itemize}
\end{frame}
\begin{frame}{Jacobian matrix computation}
\textbf{Forward kinematic derivative:} Compute analytically the partial derivative of $\vectorX_{ee}$ w.r.t $\vectorQ$
\begin{align*}
	\matriceJ &= \frac{\partial \vectorX_{ee}}{\partial\vectorQ}\\
			  &=\begin{bmatrix}
			  	\frac{\partial \vectorX_{ee}}{\partial q_1} & \frac{\partial \vectorX_{ee}}{\partial q_2} & \ldots & \frac{\partial \vectorX_{ee}}{\partial q_n}
			  \end{bmatrix}
\end{align*}
\end{frame}
