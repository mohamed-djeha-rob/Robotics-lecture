\begin{frame}{Forward differential-kinematics model}
	Forward differential-kinematics model allows to compute the task-space velocity $\vectorXdot_{ee}\inR^m$ given the joint velocity $\vectorQdot\inR^n $
	\begin{equation*}
		\vectorXdot_{ee}=\matriceJ(\vectorQ)\vectorQdot
	\end{equation*}
	\begin{itemize}
		\item $\matriceJ\inR^{m\times n}$ is the Jacobian matrix, and $\vectorXdot_{ee}=\begin{bmatrix}
			\vectorV_{ee}&\vectorOmega_{ee}
		\end{bmatrix}\top$ with $\vectorV_{ee}$ is the linear end-effector velocity  and  $\vectorOmega_{ee}$ is the angular end-effector velocity, both w.r.t the base frame
		\item Jacobian $\matriceJ$ can be decomposed into $\matriceJ =  \begin{bmatrix}
			\matriceJ_{lin}\\\matriceJ_{rot}
		\end{bmatrix}$
	\end{itemize}
\end{frame}

	Forward differential-kinematics model is used for 
	\begin{itemize}
		\item \textbf{Simulation:} Construct the links motion given a sequence of joint velocity
		\item \textbf{Estimation/Measurement:} No need to use sensor for measuring  links velocity, direct mapping of joint velocity to task-space
	\end{itemize} 
\end{frame}
\begin{frame}{Jacobian matrix computation}
	There are two approaches for computing the Jacobian matrix
	\begin{enumerate}
		\item Forward kinematics derivative
		\item Recursive velocity composition
	\end{enumerate}
\end{frame}
\begin{frame}{Jacobian matrix computation}
\textbf{Forward kinematics derivative:} Compute analytically the partial derivative of $\vectorX_{ee}$ w.r.t $\vectorQ$
\begin{align*}
	\matriceJ &= \frac{\partial \vectorX_{ee}}{\partial\vectorQ}\\
			  &=\begin{bmatrix}
			  	\frac{\partial \vectorX_{ee}}{\partial q_1} & \frac{\partial \vectorX_{ee}}{\partial q_2} & \ldots & \frac{\partial \vectorX_{ee}}{\partial q_n}
			  \end{bmatrix}
\end{align*}
\end{frame}

\begin{frame}{Jacobian matrix computation}
	The Jacobian matrix is obtained as 
	\begin{equation*}
		\matriceJ=\begin{bmatrix}
			\matriceJ_{lin}\\
			\matriceJ_{rot}
		\end{bmatrix} = \begin{bmatrix}
		\left[\overline{\sigma}_1\vectorZ_1\times\right]\vectorO_{{1}\rightarrow {ee}} +{\sigma}_1\vectorZ_1 &\cdots& \left[\overline{\sigma}_n\vectorZ_n\times\right]\vectorO_{{n}\rightarrow {ee}} +{\sigma}_n\vectorZ_n \\
		\overline{\sigma}_1\vectorZ_1 & \cdots & \overline{\sigma}_n\vectorZ_n
		\end{bmatrix}
	\end{equation*}
	\only<2->{
	\textbf{Pros:}
	\begin{itemize}
		\item Suited for implementation
		\item No analytical derivative is required
		\item Adapt easily when a structural change is performed on the robot
	\end{itemize}
	\textbf{Cons:}
	\begin{itemize}
		\item When performed \textbf{by hand}, more computational steps are required
	\end{itemize}
	}
\end{frame}
\begin{frame}{Inverse differential-kinematics model}
	Inverse differential-kinematics model allows to compute joint velocity $\vectorQdot\inR^n$ given task-space velocity $\vectorXdot_{ee}\inR^m$
	\begin{itemize}
		\item If $n=m$
		\begin{equation*}
			\vectorQdot=\matriceJ\inv\vectorXdot_{ee}
		\end{equation*}
		\item If $n>m$ (flat Jacobian) %TODO: explain redundancy
		\begin{equation*}
			\vectorQdot=\matriceJ^\dagger\vectorXdot_{ee}, \ \matriceJ^\dagger=\matriceJ^\top\left(\matriceJ\matriceJ^\top\right)\inv~\text{(Right pseudo-inverse)}
		\end{equation*}
		\item If $n<m$ (tall Jacobian)
		\begin{equation*}
			\vectorQdot=\matriceJ^\dagger\vectorXdot_{ee}, \ \matriceJ^\dagger=\left(\matriceJ^\top\matriceJ\right)\inv\matriceJ^\top~\text{(Left pseudo-inverse)}
		\end{equation*}
	\end{itemize}
\end{frame}

\begin{frame}{Inverse differential-kinematics model}
	\textbf{Inverse differential-kinematics control method:}
\begin{enumerate}%[label=\arabic*.]
	\item Given a desired end-effector velocity profil for each instant $k$  designed such that the end-effector goes to a reference pose $\vectorX_{ee}^{ref}$: 
	\begin{equation*}
		\vectorXdot_{ee}^*(k)= -\lambda(\vectorX_{ee}(k)-\vectorX_{ee}^{ref})
	\end{equation*}
	\item Compute the corresponding joint velocity with the inverse differential-kinematics model:
	\begin{equation*}
		\vectorQdot(k) = \matriceJ^\dagger\vectorXdot_{ee}^*
	\end{equation*} 
	\item Integrate to obtain new joint position:
	\begin{equation*}
		\vectorQ(k+1) = \vectorQ(k) + \vectorQdot(k)
	\end{equation*}
\end{enumerate}
\end{frame}
\begin{frame}{Inverse differential-kinematics model}
	\begin{enumerate}
		 \setcounter{enumi}{3}
	\item Use forward kinematics model to update the end effector pose:
	\begin{equation*}
		\vectorX_{ee}(k+1)\rightarrow\matriceT_{ee}^0(\vectorQ(k+1))
	\end{equation*}
	\item Loop at (1) for $k=k+1$
\end{enumerate}
\end{frame}
\begin{frame}{Kinematic redundancy}
	Robot is said to be \textbf{redundant} if it has more DoF $n$ than the task to be acheived.  \\ 
	Example: RRR robot. Task: Keep the end-effector fixed $\vectorV_{ee}=0$
	\begin{figure}
		\centering
		\includegraphics[width=0.95\columnwidth]{3DOF-Redundancy2}
		\caption{Three different possible posture resulting from the required task}
	\end{figure}
	In such case, the inverse differential kinematics solution $\vectorQdot=\matriceJ^\dagger\vectorXdot_{ee}$ \textbf{is not unique}. The extra DoF can exploited to perform a secondary task!
\end{frame}
\begin{frame}{Kinematic redundancy}
	A more general solution to the inverse differential-kinematics problem is 
	\begin{equation*}
		\vectorQdot=\matriceJ^\dagger\vectorXdot_{ee} + \matriceP \vectorQdot_0
	\end{equation*}
	with $\matriceP = \matriceI - \matriceJ^\dagger\matriceJ$ is the Jacobian null-space projector and $\vectorQdot_0$ is an arbitrary joint velocity designed to perform another task e.g., 
	\begin{itemize}
		\item keep joint away from position limits
		\item keep joint position close to a prefered posture
	\end{itemize} 
\end{frame}
\begin{frame}{Kinematic singularity}
	\begin{itemize}
		\item The Jacobian may not by inversible depending on the configuration $\vectorQ$
		\item The configurations $\vectorQ$ such that $\det(\matriceJ)=0$ are called \textbf{singular configurations}, and $\matriceJ$ is rank-deficient
		\item \textbf{Implications of singular configurations:}
		\begin{itemize}
			\item \textbf{Forward differential kinamtics:} Mobility reduction: task-space velocity is null along a certain direction
			\item \textbf{Inverse differential kinamtics:} small end-effector velocities generate infinite joint-space velocity in the neighbourhood of   singular configurations
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Kinematic singularity}
	An alternative solution overcoming the problem of inverting diﬀerential
	kinematics in the neighbourhood of a singularity is the
	damped least-squares (DLS) inverse
	\begin{equation*}
		\matriceJ^\dagger = \matriceJ^\top\left(\matriceJ\matriceJ^\top+k^2\matriceI\right)
	\end{equation*}
	with $k$ is a damping factor
\end{frame}
\begin{frame}{Second-order forward differential-kinematics model}
	Second-order forward differential-kinematics model allows to compute the task-space acceleration $\vectorXddot_{ee}\inR^m$ given the joint acceleration $\vectorQddot\inR^n $
	\begin{equation*}
		\vectorXddot_{ee}=\matriceJ(\vectorQ)\vectorQddot + \matriceJdot(\vectorQ,\vectorQdot)\vectorQdot
	\end{equation*}
	\begin{itemize}
		\item $\matriceJ(\vectorQ)\vectorQddot$ is often called the linear acceleration, $\matriceJdot(\vectorQ,\vectorQdot)\vectorQdot$ is denoted as the normal acceleration 
		\item Jacobian time-derivative can be computed analytically as $\matriceJdot(\vectorQ,\vectorQdot)= \pfrac{\matriceJ}{\vectorQ}\vectorQdot$, or numerically  $\matriceJdot(\vectorQ,\vectorQdot)=\frac{\matriceJ(\vectorQ+\vectorQdot dt) - \matriceJ(\vectorQ)}{dt}$
%		$\vectorXdot_{ee}=\begin{bmatrix}
%			\vectorV_{ee}&\vectorOmega_{ee}

%		\item Jacobian $\matriceJ$ can be decomposed into $\matriceJ =  \begin{bmatrix}
%			\matriceJ_{lin}\\\matriceJ_{rot}
%		\end{bmatrix}$
	\end{itemize}
\end{frame}
\begin{frame}{Second-order inverse differential-kinematics model}
	Second-order inverse differential-kinematics model allows to compute the joint acceleration $\vectorQddot\inR^n $ fiven the task-space acceleration $\vectorXddot_{ee}\inR^m$ 
	\begin{itemize}
		\item If $n=m$
		\begin{equation*}
			\vectorQddot=\matriceJ\inv\left(\vectorXddot_{ee}-\matriceJdot\vectorQdot\right)
		\end{equation*}
		\item If $n>m$ (flat Jacobian)
		\begin{equation*}
			\vectorQddot=\matriceJ^\dagger\left(\vectorXddot_{ee}-\matriceJdot\vectorQdot\right), \ \matriceJ^\dagger=\matriceJ^\top\left(\matriceJ\matriceJ^\top\right)\inv~\text{(Right pseudo-inverse)}
		\end{equation*}
		\item If $n<m$ (tall Jacobian)
		\begin{equation*}
			\vectorQddot=\matriceJ^\dagger\left(\vectorXddot_{ee}-\matriceJdot\vectorQdot\right), \ \matriceJ^\dagger=\left(\matriceJ^\top\matriceJ\right)\inv\matriceJ^\top~\text{(Left pseudo-inverse)}
		\end{equation*}
	\end{itemize}
\end{frame}
